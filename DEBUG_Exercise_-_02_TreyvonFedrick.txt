// This pseudocode is intended to display employee net pay values.
// All employees have a standard $45 deduction from their checks.
// If an employee does not earn enough to cover the deduction
// an error message is displayed.
// CORRECTED VERSION
start
   Declarations
      string name
      num hours
      num rate
      num DEDUCTION = 45                    // FIX: was string; it is used in math so it must be num
      string EOFNAME = "ZZZ"
      num gross
      num net

   output "Enter first name or ", EOFNAME, " to quit"
   input name

   while name <> EOFNAME                    // FIX: was "if"; a loop is needed to process many employees
      output "Enter hours worked for ", name
      input hours
      output "Enter hourly rate for ", name
      input rate
      gross = hours * rate
      net = gross - DEDUCTION
      if net > 0 then                       // FIX: was "while"; this is a decision, not a loop
         output "Net pay for ", name, " is ", net
      else
         output "Deductions not covered. Net is 0."
      endif                                 // FIX: was "endwhile"; must match the if
      output "Enter next name or ", EOFNAME, " to quit"
      input name
   endwhile                                 // FIX: was "endif"; must match the while

   output "End of job"
stop
